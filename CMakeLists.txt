cmake_minimum_required(VERSION 3.20)
project(meta_security_agent VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "MVP-0 currently targets Windows only")
endif()
add_executable(meta-sec src/main.cpp)
target_compile_definitions(meta-sec PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(meta-sec PRIVATE ole32 oleaut32 Wscapi)
if(MSVC)
  target_compile_options(meta-sec PRIVATE /W4 /EHsc)
endif()
